### Tailwind Configuration

* Extend `tailwind.config.js` with brand colors and custom background animations:

```js
module.exports = {
  darkMode: 'class',
  theme: {
    extend: {
      colors: {
        primary: '#7B61FF',
        secondary: '#FF5E5B',
        black: '#000000',
        white: '#FFFFFF',
      },
      keyframes: {
        gradientShift: {
          '0%, 100%': { 'background-position': '0% 50%' },
          '50%':     { 'background-position': '100% 50%' },
        },
        colorFade: {
          '0%':   { background: '#7B61FF' },
          '25%':  { background: '#000000' },
          '50%':  { background: '#FFFFFF' },
          '75%':  { background: '#FF5E5B' },
          '100%': { background: '#7B61FF' },
        },
      },
      animation: {
        'gradient-animate': 'gradientShift 15s ease infinite',
        'color-fade': 'colorFade 20s ease infinite',
      },
      backgroundImage: theme => ({
        'dynamic-gradient': 'linear-gradient(270deg, #7B61FF, #000000, #FFFFFF, #FF5E5B)',
      }),
    }
  },
  plugins: [require('@tailwindcss/forms')],
}
```
